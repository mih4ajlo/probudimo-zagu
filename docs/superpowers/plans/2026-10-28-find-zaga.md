`src/hint.ts`:

```ts
import { HINT_NEAR, hintFar, hintMid, MID_DISTANCE, NEAR_DISTANCE } from "./data/hunt";
import type { Point } from "./geo";

export type Compass =
  | "east"
  | "north-east"
  | "north"
  | "north-west"
  | "west"
  | "south-west"
  | "south"
  | "south-east";

// Counter-clockwise from east, 45 degrees apart.
const COMPASS: readonly Compass[] = [
  "east",
  "north-east",
  "north",
  "north-west",
  "west",
  "south-west",
  "south",
  "south-east",
];

export function compassDirection(from: Point, to: Point): Compass {
  // Game y grows south, so north is -y.
  const angle = Math.atan2(from.y - to.y, to.x - from.x);
  const sector = Math.round(angle / (Math.PI / 4));
  return COMPASS[(sector + 8) % 8];
}

export type DistanceBand = "near" | "mid" | "far";

export function distanceBand(distance: number): DistanceBand {
  if (distance < NEAR_DISTANCE) {
    return "near";
  }
  return distance < MID_DISTANCE ? "mid" : "far";
}

// What an empty office says about where she sleeps.
export function hintFor(from: Point, to: Point): string {
  const band = distanceBand(Math.hypot(to.x - from.x, to.y - from.y));
  if (band === "near") {
    return HINT_NEAR;
  }
  const direction = compassDirection(from, to);
  return band === "mid" ? hintMid(direction) : hintFar(direction);
}
```

```ts
import {
  FLEE_LINE,
  MAX_FOLLOWERS,
  MIN_HIDE_DISTANCE,
  MUMBLES,
  START_FOLLOWERS,
  WAKE_BASE,
  WAKE_FULL,
  WAKE_PER_FOLLOWER,
} from "./data/hunt";
import type { Point } from "./geo";
import { hintFor } from "./hint";

// Random number in [0, 1); scenes pass Math.random, tests a fixed value.
export type Rng = () => number;

export type HuntState = {
  positions: ReadonlyMap<string, Point>;
  rng: Rng;
  // The office she sleeps in.
  hidden: string;
  wake: number;
  followers: number;
  visits: number;
  // Offices visited at least once this run; each grows the crowd once.
  seen: string[];
  // Empty offices searched since she last moved ("?" marks).
  searched: string[];
};

export type HuntVisit = { found: boolean; lines: string[] };

const distance = (a: Point, b: Point) => Math.hypot(a.x - b.x, a.y - b.y);

export function wakeGain(followers: number): number {
  return WAKE_BASE + WAKE_PER_FOLLOWER * followers;
}

// A random office far enough from `from`, never `exclude`; any other office
// when none is far enough.
export function pickHideout(
  positions: ReadonlyMap<string, Point>,
  from: Point,
  exclude: string | null,
  rng: Rng,
): string {
  const others = [...positions.keys()].filter((id) => id !== exclude);
  const far = others.filter((id) => distance(positions.get(id) as Point, from) >= MIN_HIDE_DISTANCE);
  const pool = far.length > 0 ? far : others;
  return pool[Math.min(pool.length - 1, Math.floor(rng() * pool.length))];
}

export function createHunt(positions: ReadonlyMap<string, Point>, start: Point, rng: Rng): HuntState {
  return {
    positions,
    rng,
    hidden: pickHideout(positions, start, null, rng),
    wake: 0,
    followers: START_FOLLOWERS,
    visits: 0,
    seen: [],
    searched: [],
  };
}

export function isAwake(state: HuntState): boolean {
  return state.wake >= WAKE_FULL;
}

// Followers plus the leader.
export function crowdSize(state: HuntState): number {
  return state.followers + 1;
}

export function mumbleFor(wake: number): string {
  return (MUMBLES.find((m) => wake <= m.upTo) ?? MUMBLES[MUMBLES.length - 1]).line;
}

// The crowd reaches an office: the lines that follow the shout, or null when
// nothing happens (unknown office, or she is already awake).
export function huntVisit(state: HuntState, id: string): HuntVisit | null {
  const pos = state.positions.get(id);
  if (!pos || isAwake(state)) {
    return null;
  }
  state.visits += 1;
  if (!state.seen.includes(id)) {
    state.seen.push(id);
    state.followers = Math.min(MAX_FOLLOWERS, state.followers + 1);
  }
  if (id !== state.hidden) {
    if (!state.searched.includes(id)) {
      state.searched.push(id);
    }
    return { found: false, lines: [hintFor(pos, state.positions.get(state.hidden) as Point)] };
  }
  state.wake = Math.min(WAKE_FULL, state.wake + wakeGain(state.followers));
  state.searched = [];
  const lines = [mumbleFor(state.wake)];
  if (!isAwake(state)) {
    state.hidden = pickHideout(state.positions, pos, id, state.rng);
    lines.push(FLEE_LINE);
  }
  return { found: true, lines };
}
```

`src/crowd.ts`:

```ts
import { MAX_FOLLOWERS } from "./data/hunt";
import type { Point } from "./geo";

// Followers walk where the leader walked, this many game units apart.
export const FOLLOWER_SPACING = 9;
// The leader's path is kept as points at least this far apart.
export const TRAIL_STEP = 2;
// Path kept behind the newest point: enough for the last follower.
const KEEP_LENGTH = (MAX_FOLLOWERS + 1) * FOLLOWER_SPACING;

// The leader's recent path, newest point first.
export type Trail = { points: Point[] };

const distance = (a: Point, b: Point) => Math.hypot(a.x - b.x, a.y - b.y);

export function createTrail(start: Point): Trail {
  return { points: [{ ...start }] };
}

export function pushTrail(trail: Trail, leader: Point): void {
  if (distance(leader, trail.points[0]) < TRAIL_STEP) {
    return;
  }
  trail.points.unshift({ ...leader });
  let length = 0;
  for (let i = 1; i < trail.points.length; i++) {
    length += distance(trail.points[i - 1], trail.points[i]);
    if (length > KEEP_LENGTH) {
      trail.points.length = i + 1;
      return;
    }
  }
}

// Follower i stands i * spacing along the path behind the leader, or at the
// oldest point while the path is still shorter than that.
export function followerPositions(
  trail: Trail,
  leader: Point,
  count: number,
  spacing = FOLLOWER_SPACING,
): Point[] {
  const path = [leader, ...trail.points];
  const spots: Point[] = [];
  let segment = 0;
  // Path length from the leader to path[segment].
  let walked = 0;
  for (let i = 1; i <= count; i++) {
    const wanted = i * spacing;
    while (segment < path.length - 1 && walked + distance(path[segment], path[segment + 1]) < wanted) {
      walked += distance(path[segment], path[segment + 1]);
      segment += 1;
    }
    if (segment >= path.length - 1) {
      spots.push({ ...path[path.length - 1] });
      continue;
    }
    const a = path[segment];
    const b = path[segment + 1];
    const t = (wanted - walked) / distance(a, b);
    spots.push({ x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t });
  }
  return spots;
}
```

(A zero-length segment cannot reach the interpolation: `walked + 0 < wanted` always advances past it.)

    });
    const refresh = () => {
      options.forEach((label, i) => {
        label.text = i === selected ? `> ${MODES[i].label} <` : MODES[i].label;
        label.color = i === selected ? k.rgb(250, 220, 140) : k.rgb(150, 140, 120);
      });
    };
    const start = (i: number) => {
      resetSession();
      k.go(MODES[i].scene);
    };
    for (const [keys, delta] of [[["up", "w"], -1], [["down", "s"], 1]] as const) {
      for (const key of keys) {
        k.onKeyPress(key, () => {
          selected = moveSelection(selected, delta, MODES.length);
          refresh();
        });
      }
    }
    k.onKeyPress("enter", () => start(selected));
    refresh();
